cmake_minimum_required(VERSION 3.16)
project(danlayer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Gui)
find_package(LayerShellQt REQUIRED)
add_library(danlayer SHARED danlayer.cpp)
target_link_libraries(danlayer PRIVATE Qt6::Gui LayerShellQt::Interface)
# next to the Qt libraries it was built for (PyQt6/Qt6/lib in the bundle)
set_target_properties(danlayer PROPERTIES BUILD_RPATH_USE_ORIGIN ON INSTALL_RPATH "$ORIGIN")
install(TARGETS danlayer LIBRARY DESTINATION lib)
